from django.contrib.auth.models import UserManager


class UsuarioManager(UserManager):
    """Manager que garante o perfil de administrador ao criar superusuários."""

    def create_superuser(self, username, email=None, password=None, **extra_fields):
        extra_fields.setdefault("perfil", "administrador")
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)

        if extra_fields.get("is_staff") is not True:
            raise ValueError("Superusuário precisa ter is_staff=True.")
        if extra_fields.get("is_superuser") is not True:
            raise ValueError("Superusuário precisa ter is_superuser=True.")

        return super().create_superuser(username, email, password, **extra_fields)

from django.contrib.auth.models import AbstractUser
from django.db import models

from .managers import UsuarioManager


class Usuario(AbstractUser):
    class Perfil(models.TextChoices):
        CLIENTE = "cliente", "Cliente"
        BARBEIRO = "barbeiro", "Barbeiro"
        ADMINISTRADOR = "administrador", "Administrador/proprietário"

    perfil = models.CharField(
        "perfil",
        max_length=20,
        choices=Perfil.choices,
        default=Perfil.CLIENTE,
    )
    telefone = models.CharField(
        "telefone",
        max_length=20,
        unique=True,
        null=True,
        blank=True,
    )

    objects = UsuarioManager()

    class Meta:
        verbose_name = "usuário"
        verbose_name_plural = "usuários"
        ordering = ("username",)

    def __str__(self):
        return self.get_full_name() or self.username

    @property
    def is_cliente(self) -> bool:
        return self.perfil == self.Perfil.CLIENTE

    @property
    def is_barbeiro(self) -> bool:
        return self.perfil == self.Perfil.BARBEIRO

    @property
    def is_administrador(self) -> bool:
        return self.perfil == self.Perfil.ADMINISTRADOR or self.is_superuser

class RegistroSerializer(serializers.ModelSerializer):
    telefone = serializers.CharField(
        required=True,
        validators=[
            UniqueValidator(
                queryset=Usuario.objects.all(),
                message="Já existe um usuário com este telefone.",
            )
        ],
    )
    password = serializers.CharField(write_only=True, min_length=8)
    password_confirm = serializers.CharField(write_only=True)

    class Meta:
        model = Usuario
        fields = (
            "id",
            "username",
            "email",
            "first_name",
            "last_name",
            "telefone",
            "password",
            "password_confirm",
        )
        read_only_fields = ("id",)

    def validate_email(self, value):
        if value and Usuario.objects.filter(email__iexact=value).exists():
            raise serializers.ValidationError("Já existe um usuário com este e-mail.")
        return value

    def validate(self, attrs):
        if attrs["password"] != attrs["password_confirm"]:
            raise serializers.ValidationError(
                {"password_confirm": "As senhas não conferem."}
            )
        return attrs

    def create(self, validated_data):
        validated_data.pop("password_confirm")
        password = validated_data.pop("password")
        return Usuario.objects.create_user(
            perfil=Usuario.Perfil.CLIENTE,
            password=password,
            **validated_data,
        )


class LoginSerializer(TokenObtainPairSerializer):

        data = super().validate(attrs)
        data["user"] = UsuarioSerializer(self.user).data
        return data

from rest_framework import generics, status
from rest_framework.permissions import AllowAny, IsAuthenticated
from rest_framework.response import Response
from rest_framework.views import APIView
from rest_framework_simplejwt.exceptions import TokenError
from rest_framework_simplejwt.tokens import RefreshToken
from rest_framework_simplejwt.views import TokenObtainPairView

from .serializers import (
    LoginSerializer,
    LogoutSerializer,
    RegistroSerializer,
    UsuarioSerializer,
)


class RegisterView(generics.CreateAPIView):
    permission_classes = [AllowAny]
    serializer_class = RegistroSerializer

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        usuario = serializer.save()

        refresh = RefreshToken.for_user(usuario)
        return Response(
            {
                "user": UsuarioSerializer(usuario).data,
                "access": str(refresh.access_token),
                "refresh": str(refresh),
            },
            status=status.HTTP_201_CREATED,
        )


class LoginView(TokenObtainPairView):
    permission_classes = [AllowAny]
    serializer_class = LoginSerializer


class LogoutView(APIView):
    permission_classes = [IsAuthenticated]

    def post(self, request):
        serializer = LogoutSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        try:
            RefreshToken(serializer.validated_data["refresh"]).blacklist()
        except TokenError:
            return Response(
                {"detail": "Token inválido ou já expirado."},
                status=status.HTTP_400_BAD_REQUEST,
            )
        return Response(status=status.HTTP_204_NO_CONTENT)


class MeView(generics.RetrieveUpdateAPIView):
    permission_classes = [IsAuthenticated]
    serializer_class = UsuarioSerializer

    def get_object(self):
        return self.request.user

## 11. Modelo de dados (MVP)

```
Barbearia (registro único)             # catalog
  proprietario  -> FK Usuario (perfil administrador)
  nome, descricao, endereco, cidade, telefone, email

Usuario(AbstractUser)                 # accounts
  perfil        -> cliente | barbeiro | administrador
  telefone
  is_staff/is_superuser (Admin)

Profissional                          # catalog
  usuario       -> FK Usuario (perfil barbeiro)
  especialidades, bio, ativo
  (1 profissional por usuário)

Servico                               # catalog
  nome, descricao, duracao_minutos, preco, ativo

Disponibilidade                       # scheduling
  profissional  -> FK Profissional
  dia_semana    -> 0..6
  hora_inicio, hora_fim
  (vários blocos por dia permitidos)

Agendamento                           # scheduling
  cliente       -> FK Usuario (perfil cliente)
  profissional  -> FK Profissional
  servico       -> FK Servico
  data_hora     -> datetime (timezone tratado na aplicação: America/Sao_Paulo)
  duracao_minutos (copiada do serviço)
  preco         (copiada do serviço)
  status        -> pendente | confirmado | concluido | cancelado
  observacoes
  criado_em / atualizado_em
```

Índices e constraints previstos:
- `Barbearia` com apenas um registro (singleton).
- `Usuario.telefone` único.
- Índices em `profissional_id`, `cliente_id`, `data_hora` e `status`.
- Índice composto em (`profissional`, `data_hora`) para consultas de agenda e conflito.
- MySQL não possui exclusion constraint: a sobreposição de horários é impedida por
  transação com bloqueio (`SELECT ... FOR UPDATE`) em `scheduling/services.py`.
- Testes automatizados cobrindo concorrência e conflito de horários.

Regras de negócio:
- Horário deve estar dentro de um bloco de `Disponibilidade` do profissional.
- Não pode haver dois agendamentos ativos (`pendente`/`confirmado`) sobrepostos para o mesmo
  profissional.
- `data_hora` não pode estar no passado.
- A duração do serviço ocupa o slot; o próximo horário só libera após o término.
- Ao cancelar, o horário volta a ficar disponível.
- Preço e duração são copiados no agendamento para preservar o histórico.
- Cliente só acessa os próprios agendamentos.
- Barbeiro só acessa a própria agenda.
- Administrador/proprietário gerencia a operação da barbearia.

## 16. Riscos e decisões

|---|---|
| Conflito de horários | Transação com lock (`SELECT ... FOR UPDATE`) + índice composto + testes; MySQL não possui exclusion constraint |
| Segurança dos tokens JWT | Access token curto, refresh com blacklist e rotação |
| Dependência de MySQL externo | Credenciais, disponibilidade e permissões acordadas com o responsável pelo banco; migrations testadas no ambiente fornecido |

| Serviço mudar de preço e alterar histórico | Copiar preço/duração no agendamento |
| Escopo crescer | MVP limitado aos 6 módulos e sem extras |
| Necessidade futura de múltiplas unidades | Modelo atual é de uma barbearia; migração futura exigirá análise |

## Regras de negócio

- **Perfis:** `cliente`, `barbeiro`, `administrador`. O superusuário do Django é tratado
  como administrador.
- **Uma barbearia:** `Barbearia` é um registro único (pk=1), sem isolamento multi-tenant.
- **Preço e duração** são copiados do serviço no momento da reserva, preservando o histórico.
- **Status:** `pendente → confirmado → concluido` ou `pendente/confirmado → cancelado`.
  Transições inválidas retornam `400`.
- **Horário válido:** precisa estar dentro de um bloco de `Disponibilidade` do profissional,
  no futuro e sem sobreposição com outro agendamento ativo.
- **Slots:** gerados a cada 30 minutos; a duração do serviço ocupa o intervalo e o horário
  só é liberado após o término.
- **Conflitos:** o MySQL não possui exclusion constraint. A criação do agendamento usa
  `transaction.atomic()` + `SELECT ... FOR UPDATE` no profissional, com índice composto em
  `(profissional, data_hora)`.
- **Cancelamento** libera o horário para novas reservas.
- **Fuso horário:** datas são armazenadas em UTC e expostas em `America/Sao_Paulo`.

            defaults=campos,

"""Regras de negócio de disponibilidade e agendamento.

A prevenção de conflito usa transação + ``SELECT ... FOR UPDATE`` no registro do
profissional. O MySQL não possui exclusion constraint, então o bloqueio da linha
serializa as reservas concorrentes do mesmo profissional.
"""
from datetime import datetime, time, timedelta

from django.db import transaction
from django.utils import timezone
from rest_framework.exceptions import ValidationError

from catalog.models import Profissional

from .models import Agendamento, Disponibilidade

INTERVALO_SLOT_MINUTOS = 30

STATUS_ATIVOS = (
    Agendamento.Status.PENDENTE,
    Agendamento.Status.CONFIRMADO,
)

TRANSICOES_VALIDAS = {
    Agendamento.Status.PENDENTE: {
        Agendamento.Status.CONFIRMADO,
        Agendamento.Status.CANCELADO,
    },
    Agendamento.Status.CONFIRMADO: {
        Agendamento.Status.CONCLUIDO,
        Agendamento.Status.CANCELADO,
    },
    Agendamento.Status.CONCLUIDO: set(),
    Agendamento.Status.CANCELADO: set(),
}


def normalizar_inicio(valor: datetime) -> datetime:
    """Garante um datetime consciente de fuso no fuso local do projeto."""
    if timezone.is_naive(valor):
        return timezone.make_aware(valor, timezone.get_current_timezone())
    return timezone.localtime(valor)


def _aware(data, hora) -> datetime:
    return timezone.make_aware(
        datetime.combine(data, hora),
        timezone.get_current_timezone(),
    )


def _sobrepoe(inicio_a, fim_a, inicio_b, fim_b) -> bool:
    return inicio_a < fim_b and inicio_b < fim_a


def _intervalo_do_dia(data):
    inicio = _aware(data, time.min)
    return inicio, inicio + timedelta(days=1)


def _agendamentos_do_dia(profissional, data, excluir_pk=None):
    inicio, fim = _intervalo_do_dia(data)
    qs = Agendamento.objects.filter(
        profissional=profissional,
        status__in=STATUS_ATIVOS,
        data_hora__gte=inicio,
        data_hora__lt=fim,
    )
    if excluir_pk:
        qs = qs.exclude(pk=excluir_pk)
    return list(qs)


def existe_conflito(profissional, inicio, fim, excluir_pk=None) -> bool:
    """Verifica sobreposição com agendamentos ativos do profissional."""
    data = timezone.localtime(inicio).date()
    for agendamento in _agendamentos_do_dia(profissional, data, excluir_pk):
        if _sobrepoe(inicio, fim, agendamento.data_hora, agendamento.fim):
            return True
    return False


def esta_dentro_da_disponibilidade(profissional, inicio, fim) -> bool:
    """Confere se todo o intervalo cabe em um bloco semanal do profissional."""
    inicio_local = timezone.localtime(inicio)
    fim_local = timezone.localtime(fim)
    if inicio_local.date() != fim_local.date():
        return False

    hora_inicio = inicio_local.time()
    hora_fim = fim_local.time()
    blocos = Disponibilidade.objects.filter(
        profissional=profissional,
        dia_semana=inicio_local.weekday(),
        ativo=True,
    )
    return any(
        bloco.hora_inicio <= hora_inicio and hora_fim <= bloco.hora_fim
        for bloco in blocos
    )


def horarios_disponiveis(
    profissional,
    data,
    duracao_minutos: int = INTERVALO_SLOT_MINUTOS,
):
    """Retorna a lista de horários livres do profissional em uma data."""
    duracao = timedelta(minutes=duracao_minutos)
    blocos = Disponibilidade.objects.filter(
        profissional=profissional,
        dia_semana=data.weekday(),
        ativo=True,
    ).order_by("hora_inicio")

    if not blocos:
        return []

    agora = timezone.now()
    ocupados = [
        (agendamento.data_hora, agendamento.fim)
        for agendamento in _agendamentos_do_dia(profissional, data)
    ]

    livres = []
    for bloco in blocos:
        candidato = _aware(data, bloco.hora_inicio)
        fim_bloco = _aware(data, bloco.hora_fim)
        while candidato + duracao <= fim_bloco:
            fim_candidato = candidato + duracao
            ocupado = any(
                _sobrepoe(candidato, fim_candidato, inicio, fim)
                for inicio, fim in ocupados
            )
            if candidato > agora and not ocupado:
                livres.append(candidato)
            candidato += timedelta(minutes=INTERVALO_SLOT_MINUTOS)

    livres.sort()
    unicos = []
    for slot in livres:
        if not unicos or unicos[-1] != slot:
            unicos.append(slot)
    return unicos


def criar_agendamento(*, cliente, profissional, servico, inicio, observacoes=""):
    """Cria um agendamento validando disponibilidade e conflito de forma atômica."""
    inicio = normalizar_inicio(inicio)

    if not cliente.is_cliente:
        raise ValidationError(
            {"detail": "Apenas clientes podem criar agendamentos."}
        )
    if not servico.ativo:
        raise ValidationError({"servico": "Serviço indisponível."})
    if inicio <= timezone.now():
        raise ValidationError(
            {"data_hora": "Não é possível agendar em data/hora passada."}
        )

    fim = inicio + timedelta(minutes=servico.duracao_minutos)

    with transaction.atomic():
        # Bloqueia o profissional para serializar reservas concorrentes.
        profissional = Profissional.objects.select_for_update().get(
            pk=profissional.pk
        )
        if not profissional.ativo:
            raise ValidationError({"profissional": "Profissional indisponível."})
        if not esta_dentro_da_disponibilidade(profissional, inicio, fim):
            raise ValidationError(
                {
                    "data_hora": (
                        "Horário fora da disponibilidade do profissional."
                    )
                }
            )
        if existe_conflito(profissional, inicio, fim):
            raise ValidationError({"data_hora": "Horário já está ocupado."})

        return Agendamento.objects.create(
            cliente=cliente,
            profissional=profissional,
            servico=servico,
            data_hora=inicio,
            duracao_minutos=servico.duracao_minutos,
            preco=servico.preco,
            observacoes=observacoes,
        )


def atualizar_status(agendamento, novo_status):
    """Aplica uma transição de status válida."""
    if novo_status == agendamento.status:
        return agendamento

    permitidos = TRANSICOES_VALIDAS.get(agendamento.status, set())
    if novo_status not in permitidos:
        raise ValidationError(
            {
                "status": (
                    f"Transição de '{agendamento.get_status_display()}' para "
                    f"'{dict(Agendamento.Status.choices).get(novo_status, novo_status)}' "
                    "não permitida."
                )
            }
        )

    agendamento.status = novo_status
    agendamento.save(update_fields=["status", "atualizado_em"])
    return agendamento


def cancelar_agendamento(agendamento):
    """Cancela um agendamento pendente ou confirmado, liberando o horário."""
    return atualizar_status(agendamento, Agendamento.Status.CANCELADO)

SELECT_RELATED = ("cliente", "profissional__usuario", "servico")


def _intervalo_do_dia(data):
    inicio = timezone.make_aware(
        datetime.combine(data, time.min),
        timezone.get_current_timezone(),
    )
    return inicio, inicio + timedelta(days=1)


def _data_ou_hoje(valor):
    if not valor:
        return timezone.localdate()
    data = parse_date(valor)
    if data is None:
        raise ValidationError(
            {"date": "Data inválida. Use o formato AAAA-MM-DD."}
        )
    return data


def _profissional_do_usuario(usuario):
    try:
        return usuario.profissional
    except Profissional.DoesNotExist:
        return None


def _id_valido(valor, campo):
    if valor in (None, ""):
        return None
    if not str(valor).isdigit():
        raise ValidationError({campo: "Informe um ID numérico válido."})
    return int(valor)


def _aplicar_filtros(qs, request):
    status_param = request.query_params.get("status")
    if status_param:
        if status_param not in Agendamento.Status.values:
            raise ValidationError({"status": "Status inválido."})
        qs = qs.filter(status=status_param)

    de = request.query_params.get("from")
    if de:
        data = parse_date(de)
        if data is None:
            raise ValidationError(
                {"from": "Data inválida. Use o formato AAAA-MM-DD."}
            )
        qs = qs.filter(data_hora__gte=_intervalo_do_dia(data)[0])

    ate = request.query_params.get("to")
    if ate:
        data = parse_date(ate)
        if data is None:
            raise ValidationError(
                {"to": "Data inválida. Use o formato AAAA-MM-DD."}
            )
        qs = qs.filter(data_hora__lt=_intervalo_do_dia(data)[1])

    return qs


def _agenda_queryset(usuario):
    qs = Agendamento.objects.select_related(*SELECT_RELATED)
    if usuario.is_barbeiro:
        profissional = _profissional_do_usuario(usuario)
        if profissional is None:
            raise PermissionDenied(
                "Usuário barbeiro sem perfil profissional cadastrado."
            )
        return qs.filter(profissional=profissional)
    if usuario.is_administrador:
        return qs
    raise PermissionDenied("Acesso restrito a barbeiros e administradores.")


class ProfissionalDisponibilidadeView(APIView):
    """Horários livres do profissional por data ou por período."""

    permission_classes = [AllowAny]

    def get(self, request, pk):
        profissional = get_object_or_404(
            Profissional.objects.filter(
                ativo=True,
                usuario__is_active=True,
            ),
            pk=pk,
        )

        if request.query_params.get("date"):
            data = _data_ou_hoje(request.query_params.get("date"))
            duracao = INTERVALO_SLOT_MINUTOS
            servico_id = _id_valido(
                request.query_params.get("service"),
                "service",
            )
            if servico_id:
                servico = Servico.objects.filter(
                    pk=servico_id,
                    ativo=True,
                ).first()
                if servico is None:
                    raise ValidationError(
                        {"service": "Serviço não encontrado."}
                    )
                duracao = servico.duracao_minutos
            slots = horarios_disponiveis(profissional, data, duracao)
            return Response(
                {
                    "date": data.isoformat(),
                    "duration_minutes": duracao,
                    "slots": [slot.isoformat() for slot in slots],
                }
            )

        de = parse_date(request.query_params.get("from") or "")
        ate = parse_date(request.query_params.get("to") or "")
        if de is None or ate is None:
            raise ValidationError(
                {
                    "from": (
                        "Informe 'date' ou o período com 'from' e 'to' "
                        "(AAAA-MM-DD)."
                    )
                }
            )
        if ate < de:
            raise ValidationError(
                {"to": "A data final deve ser maior ou igual à inicial."}
            )
        if (ate - de).days > 60:
            raise ValidationError(
                {"to": "O período máximo consultado é de 60 dias."}
            )

        dias = []
        atual = de
        while atual <= ate:
            slots = horarios_disponiveis(profissional, atual)
            if slots:
                dias.append(
                    {
                        "date": atual.isoformat(),
                        "slots": [slot.isoformat() for slot in slots],
                    }
                )
            atual += timedelta(days=1)

        return Response(
            {"from": de.isoformat(), "to": ate.isoformat(), "days": dias}
        )


class MeusAgendamentosView(generics.ListCreateAPIView):
    permission_classes = [IsCliente]

    def get_queryset(self):
        return _aplicar_filtros(
            Agendamento.objects.filter(
                cliente=self.request.user
            ).select_related(*SELECT_RELATED),
            self.request,
        )

    def get_serializer_class(self):
        if self.request.method == "POST":
            return AgendamentoCreateSerializer
        return AgendamentoSerializer

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        agendamento = serializer.save()
        return Response(
            AgendamentoSerializer(agendamento).data,
            status=status.HTTP_201_CREATED,
        )


class AgendamentoDetailView(generics.RetrieveAPIView):
    permission_classes = [IsAuthenticated]
    serializer_class = AgendamentoSerializer

    def get_queryset(self):
        usuario = self.request.user
        qs = Agendamento.objects.select_related(*SELECT_RELATED)
        if usuario.is_administrador:
            return qs
        if usuario.is_barbeiro:
            profissional = _profissional_do_usuario(usuario)
            if profissional is None:
                return qs.none()
            return qs.filter(profissional=profissional)
        return qs.filter(cliente=usuario)


class CancelarAgendamentoView(APIView):

    def patch(self, request, pk):
        agendamento = get_object_or_404(
            Agendamento.objects.select_related(*SELECT_RELATED),

            )
        agendamento = cancelar_agendamento(agendamento)
        return Response(AgendamentoSerializer(agendamento).data)


class MinhaAgendaView(APIView):

        qs = _agenda_queryset(request.user)

        inicio, fim = _intervalo_do_dia(data)

        return Response(AgendamentoSerializer(qs, many=True).data)


class MinhaAgendaSemanaView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        qs = _agenda_queryset(request.user)
        data = _data_ou_hoje(request.query_params.get("date"))
        inicio_semana = data - timedelta(days=data.weekday())
        inicio, _ = _intervalo_do_dia(inicio_semana)
        fim = inicio + timedelta(days=7)
        qs = qs.filter(data_hora__gte=inicio, data_hora__lt=fim)
        return Response(
            {
                "from": inicio_semana.isoformat(),
                "to": (inicio_semana + timedelta(days=6)).isoformat(),
                "appointments": AgendamentoSerializer(qs, many=True).data,
            }
        )


class AtualizarStatusAgendamentoView(APIView):
    permission_classes = [IsAuthenticated]

    def patch(self, request, pk):
        agendamento = get_object_or_404(
            Agendamento.objects.select_related(*SELECT_RELATED),
            pk=pk,
        )
        usuario = request.user
        if usuario.is_barbeiro:
            profissional = _profissional_do_usuario(usuario)
            if (
                profissional is None
                or agendamento.profissional_id != profissional.id
            ):
                raise PermissionDenied(
                    "Você só pode alterar os próprios atendimentos."
                )
        elif not usuario.is_administrador:
            raise PermissionDenied(
                "Acesso restrito a barbeiros e administradores."
            )

        serializer = AgendamentoStatusSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        agendamento = atualizar_status(
            agendamento,
            serializer.validated_data["status"],
        )
        return Response(AgendamentoSerializer(agendamento).data)
